/* Function - No Args and No return type */
#include <iostream>
using namespace std;
class Sum{
    private:
        int a, b, sum = 0;
    public:
        void input(){
            cout<<"Enter number : "<<endl;
            cin>>a>>b;
        }
        int add(){
            sum = a + b;
            return(sum);
        }
};
int main() {
     Sum s;
    s.input();
    int out = s.add();
    
    cout<<"Sum of values: "<<out;
    
    return 0;
}
